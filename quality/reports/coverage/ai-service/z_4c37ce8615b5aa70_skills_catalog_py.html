111_BOUNDARY_BEFORE = r"(?<![0-9A-Za-zÀ-ÿ+#.])" 

112_BOUNDARY_AFTER = r"(?![0-9A-Za-zÀ-ÿ+#])"

115def _pattern(alias: str) -> re.Pattern: 

116 body = re.escape(alias).replace(r"\ ", r"\s+") 

117 # Une variante écrite avec une majuscule (« Go », « JS », « REST ») n'est reconnue qu'à l'identique. 

118 flags = 0 if alias != alias.lower() else re.IGNORECASE 

119 return re.compile(f"{_BOUNDARY_BEFORE}{body}{_BOUNDARY_AFTER}", flags) 

120 

121 

122# (canonique, motif) trié par variante la plus longue d'abord, pour que « Spring Boot » 

123# soit reconnu avant « Spring ». 

124_PATTERNS: list[tuple[str, re.Pattern]] = [ 

125 (canonical, _pattern(alias)) 

126 for canonical, aliases in CATALOG.items() 

127 for alias in sorted(aliases, key=len, reverse=True) 

128] 

129 

130_KEY_TO_CANONICAL: dict[str, str] = { 

131 normalize(alias): canonical for canonical, aliases in CATALOG.items() for alias in (canonical, *aliases) 

132}

135def skill_key(skill: str) -> str: 

136 """Clé de comparaison d'une compétence : « Postgres » et « PostgreSQL » donnent la même.""" 

137 key = normalize(skill) 

138 canonical = _KEY_TO_CANONICAL.get(key) 

139 return normalize(canonical) if canonical else key

142def extract_skills(text: str) -> list[str]: 

143 """Compétences du catalogue présentes dans `text`, dans l'ordre d'apparition, sans doublon. 

144 

145 Une compétence incluse dans une autre déjà reconnue au même endroit (« Spring » dans 

146 « Spring Boot ») n'est pas comptée à part. 

147 """ 

148 found: list[tuple[int, int, str]] = [] 

149 for canonical, pattern in _PATTERNS: 

150 for match in pattern.finditer(text): 

151 found.append((match.start(), match.end(), canonical)) 

152 found.sort(key=lambda item: (item[0], -(item[1] - item[0]))) 

153 

154 result: list[str] = [] 

155 last_end = -1 

156 for start, end, canonical in found: 

157 if start < last_end: 

158 continue 

159 last_end = end 

160 if canonical not in result: 

161 result.append(canonical) 

162 return result
