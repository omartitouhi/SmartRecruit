41def _skill_overlap(cv: CVParsingResult, job: AnalyzedJob) -> int: 

42 cv_skills = {normalize(skill) for skill in cv.skills} 

43 return sum(normalize(skill) in cv_skills for skill in job.analysis.required_skills)

46def shortlist(cv: CVParsingResult, jobs: list[AnalyzedJob]) -> list[AnalyzedJob]: 

47 """Présélection déterministe : recouvrement de compétences décroissant (tri stable, 

48 les offres les plus récemment analysées d'abord à égalité).""" 

49 return sorted(jobs, key=lambda job: _skill_overlap(cv, job), reverse=True)[:MAX_CANDIDATE_JOBS]

72def keep_known_jobs(output: LLMRecommendations, jobs: list[AnalyzedJob]) -> list[Recommendation]: 

73 """Anti-hallucination : tout identifiant absent des offres fournies est rejeté ; 

74 doublons écartés, tri par score décroissant, 10 offres au plus.""" 

75 known = {str(job.job_id).lower(): job.job_id for job in jobs} 

76 best: dict = {} 

77 for item in output.recommendations: 

78 job_id = known.get(item.jobId.strip().lower()) 

79 if job_id is None: 

80 log.warning("Recommandation rejetée : offre inconnue %r renvoyée par le LLM", item.jobId) 

81 continue 

82 if job_id not in best: 

83 best[job_id] = item.score 

84 ranked = sorted(best.items(), key=lambda pair: pair[1], reverse=True) 

85 return [Recommendation(job_id=job_id, score=score) for job_id, score in ranked[:MAX_RECOMMENDATIONS]]
