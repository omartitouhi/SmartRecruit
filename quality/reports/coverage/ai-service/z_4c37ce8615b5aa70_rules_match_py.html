45 for start, end in periods:

90def match_criteria_rules(cv: CVParsingResult, job: AnalyzedJob) -> dict[str, tuple[float | None, str]]: 

91 """Sous-score (0-100, None si non évaluable) et explication de chaque critère."""

93 owned = cv_skill_keys(cv) 

94 covered = [s for s in required if skill_key(s) in owned] 

95 missing = [s for s in required if s not in covered] 

96 if required and (cv.skills or owned): 96 ↛ 107line 96 didn't jump to line 107 because the condition on line 96 was always true

97 plural = "s" if len(covered) > 1 else "" 

98 if covered: 

99 explanation = f"{len(covered)} compétence{plural} requise{plural} sur {len(required)} présente{plural} dans le CV" 

100 explanation += f" ({_list(covered)})" 

101 else: 

102 explanation = (f"Aucune des {len(required)} compétences requises n'est présente dans le CV" 

103 if len(required) > 1 else "La compétence requise n'est pas présente dans le CV") 

104 explanation += f" ; manque : {_list(missing)}." if missing else "." 

105 skills = (round(100 * len(covered) / len(required), 1), explanation) 

106 else: 

107 skills = (None, "compétences non comparables.") 

108 

109 level = job.analysis.experience_level 

110 expected = years_expected(level) 

111 if cv.experiences: 111 ↛ 118line 111 didn't jump to line 118 because the condition on line 111 was always true

112 years = experience_years(cv) 

113 score = 100.0 if years >= expected else round(100 * years / expected, 1) 

114 wanted = f"{_years_phrase(expected)} attendus" if expected else "aucune expérience exigée" 

115 experience = (score, f"{_years_phrase(years)} d'expérience cumulée pour un poste de niveau " 

116 f"{LEVEL_LABELS[level]} ({wanted}).") 

117 else: 

118 experience = (None, "aucune expérience extraite du CV.")

124 top = next(e.degree for e in cv.education if degree_rank(e.degree) == best) 

125 education = (min(100.0, round(100 * best / needed, 1)),

132def recommend_rules(cv: CVParsingResult, jobs: list[AnalyzedJob]) -> RecommendationResult: 

133 """Offres ayant au moins une compétence requise en commun, classées par le même score que le matching.""" 

134 owned = cv_skill_keys(cv) 

135 ranked = [] 

136 for job in jobs: 

137 if not any(skill_key(s) in owned for s in job.analysis.required_skills): 

138 continue 

139 sub_scores = {c: s for c, (s, _) in match_criteria_rules(cv, job).items() if s is not None} 

140 ranked.append((compute_global_score(sub_scores), job)) 

141 ranked.sort(key=lambda pair: pair[0], reverse=True) 

142 return RecommendationResult( 

143 recommendations=[Recommendation(job_id=job.job_id, score=score) for score, job in ranked[:MAX_RECOMMENDATIONS]] 

144 )
