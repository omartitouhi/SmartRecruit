58def _valid_entries(model: type[BaseModel], entries: list[dict]) -> list: 

59 """Garde-fou anti-hallucination : toute entrée non conforme au schéma est rejetée.""" 

60 result = [] 

61 for entry in entries: 

62 try: 

63 result.append(model.model_validate(entry)) 

64 except ValidationError: 

65 log.info("Entrée de CV rejetée (non conforme au schéma) : %s", entry) 

66 return result

69def _language_in_text(language: str, text: str) -> bool: 

70 pattern = LANGUAGE_ALIASES.get(language) 

71 if pattern: 

72 return re.search(rf"\b(?:{pattern})\b", text, re.IGNORECASE) is not None 

73 return appears_in(language, text)

76def ground_cv_output(output: LLMCVOutput, text: str) -> CVParsingResult: 

77 """Ne conserve que les données conformes au schéma ET présentes dans le texte source.""" 

78 experiences = [ 

79 e for e in _valid_entries(_StrictExperience, output.experiences) 

80 if e.title.strip() and appears_in(e.company, text) 

81 ] 

82 education = [ 

83 e for e in _valid_entries(_StrictEducation, output.education) 

84 if e.degree.strip() and appears_in(e.institution, text) 

85 ] 

86 return CVParsingResult( 

87 skills=dedupe([s for s in output.skills if appears_in(s, text)]), 

88 experiences=[ParsedExperience(**e.model_dump()) for e in experiences], 

89 education=[ParsedEducation(**e.model_dump()) for e in education], 

90 languages=dedupe([lang for lang in output.languages if _language_in_text(lang, text)]), 

91 )
